```ts
import { glyph, type Font, type FontFace } from '@pmndrs/glyph';
import type { Text } from '@pmndrs/glyph/three';
import { bitmap } from '@pmndrs/glyph/raster/bitmap';
import { msdf } from '@pmndrs/glyph/raster/msdf';
import { slug } from '@pmndrs/glyph/raster/slug';

const Inter = glyph.fontFace('/fonts/Inter.glb', { format: [msdf, slug, bitmap({ strikes: [16, 32] })] });
const declared: FontFace<typeof msdf | typeof slug | ReturnType<typeof bitmap>> = Inter; // the tuple became a union
const title: Text<typeof slug> = three.createText({ font: Inter.slug, text: 'Hero' }); // inferred from the member
const body: Font<typeof msdf> = inter; // what useMsdf(url) hands you
```

## Inference

| Site | Infers from |
| --- | --- |
| `glyph.fontFace(url, { format })` | the `format` value or tuple; members are keyed by each format's `kind` |
| `Inter.slug`, `Inter.bitmap` | the declared member; an undeclared key is a compile error |
| `three.createText({ font })` | a face, a member, or a loaded `Font`; a family string erases to `AnyRasterFormat` |
| `<Text font={…}>` | the same selection; `ref` is `Ref<Text<Technique>>` |
| `useMsdf` / `useSlug` / `useBitmap` | fixed: `Font<typeof msdf>`, `Font<typeof slug>`, `Font<typeof bitmap>` |
| `defineTextMaterial((context) => …)` | `context.kind`, then `context.format` |

```tsx
import { Text } from '@pmndrs/glyph/react';
import { useSlug } from '@pmndrs/glyph/react/slug';
import type { Text as ThreeText } from '@pmndrs/glyph/three';
import type { slug } from '@pmndrs/glyph/raster/slug';
import { useRef } from 'react';

export function Title() {
  const title = useSlug(url);
  const ref = useRef<ThreeText<typeof slug>>(null);
  return (
    <Text ref={ref} font={title}>
      Hero
    </Text>
  );
}
```

## Narrowing a material

```ts
import { defineTextMaterial } from '@pmndrs/glyph/three';

const soft = defineTextMaterial((context) => {
  const material = context.createDefaultMaterial();
  if (context.kind === 'glyph' && context.format === 'pmndrs.msdf') {
    material.opacityNode = context.shader.fillCoverage; // TslMsdfShaderOutput here, nothing else
  }
  return material;
});
```

{/*
write
- Question: how does a material know which format it is drawing?
- Two paragraphs: `context` is a closed discriminated union — `kind: 'decoration'` with no format, or
  `kind: 'glyph'` with `format` one of the three built-ins — and `shader` narrows with it, so the MSDF branch sees
  `fillCoverage` and the Slug branch sees `renderCoordinate`. A custom raster augments the map at its own format
  literal, which keeps the switch exhaustive without a catch-all string.
- Shape: two short paragraphs.

notes · internals

## PropertyList

```ts
type PropertyList<Value> = Value | false | null | undefined | readonly PropertyList<Value>[];
```

```ts
import { Constraints, ParagraphLayout, TextStyle } from '@pmndrs/glyph';

const styles = TextStyle.create({ base: { fontSize: 16 }, accent: [{ color: '#ffd166' }, false, null] });
const layouts = ParagraphLayout.create({ prose: { wrap: 'word', align: 'start' } });
const boxes = Constraints.create({ card: { width: { mode: 'at-most', size: 320 } } });

text.style = [styles.base, styles.accent, { opacity: 0.8 }]; // left to right; falsy entries are skipped
text.layout = layouts.prose;
text.constraints = boxes.card;
```

## Text input

```ts
import type { AnyRasterFormat, FormattedText } from '@pmndrs/glyph';

type TextInput<T extends AnyRasterFormat> = string | FormattedText<T>;
```

## Branded values

| Type | Where |
| --- | --- |
| `GlyphKey` | `GlyphPlacement.key` on placements read through `caretAt` and `measureGlyphs` |
| `RasterKey`, `Sha256Hex` | artifact identity: the raster descriptor hash, the GLB content hash |
| `FontSlot`, `LocalGlyphId` | plain numbers, exported for readability, not branded |

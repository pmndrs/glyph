## The format

```ts
import type { JsonValue, RasterDecodeArtifact, RasterDecodeFont, RasterResourceId, RasterResourceSource } from '@pmndrs/glyph';
import { defineRasterFormat, defineRasterResourceId } from '@pmndrs/glyph/config/raster-format';

export interface FrameOptions {
  /** Fraction of the em removed from each side of the frame. */
  readonly inset?: number;
}

/** The canonical descriptor: defaults folded in, so "no options" and "the defaults spelled out" are one raster. */
export interface FrameDescriptor {
  readonly [key: string]: JsonValue;
  readonly generatorVersion: '0.0.0';
  readonly inset: number;
}

export interface FrameData {
  readonly resource: RasterResourceId;
  readonly inset: number;
  readonly colors: Uint8Array; // four bytes per glyph
  readonly glyphCount: number;
}

export function frameDescriptor(options: FrameOptions = {}): FrameDescriptor {
  const inset = options.inset ?? 0.08;
  if (!Number.isFinite(inset) || inset < 0 || inset >= 0.5) throw new RangeError('frame inset must be in [0, 0.5)');
  return { generatorVersion: '0.0.0', inset };
}

/** What the companion extension carries; the example validates every field before trusting it. */
interface FrameExtension {
  readonly version: 0;
  readonly rasterKey: string;
  readonly shapingHash: string;
  readonly glyphCount: number;
  readonly descriptor: FrameDescriptor;
  readonly records: RasterResourceSource;
}

export const frame = defineRasterFormat({
  id: 'studio.frame', // wire identity; `pmndrs.` is reserved
  kind: 'frame', // the key a handle's fonts map and `face.frame` use
  extension: 'STUDIO_frame',
  version: 0,
  textEffects: [], // only 'outline' and 'shadow' are legal here
  descriptor(options: FrameOptions | undefined): FrameDescriptor {
    return frameDescriptor(options);
  },
  async decode(font: RasterDecodeFont, raster: RasterDecodeArtifact<'frame'>, signal?: AbortSignal): Promise<FrameData> {
    const extension = raster.extensionData as unknown as FrameExtension;
    if (extension.version !== 0 || extension.rasterKey !== raster.rasterKey || extension.shapingHash !== font.shapingHash) {
      throw new TypeError('frame extension identity does not match its registered font');
    }
    const colors = Uint8Array.from(await raster.resource(extension.records, signal));
    if (colors.byteLength !== font.glyphCount * 4) throw new RangeError('frame records do not cover every glyph');
    return {
      resource: defineRasterResourceId(`studio.frame/${font.shapingHash}/${raster.rasterKey}`),
      inset: extension.descriptor.inset,
      colors,
      glyphCount: font.glyphCount,
    };
  },
  dispose(data: FrameData): void {
    data.colors.fill(0);
  },
});
```

## The plan program

```ts
import type { PortableGeometryPayload, RasterFormat, RasterFormatId, RasterResourceId } from '@pmndrs/glyph';
import { id } from '@pmndrs/glyph/config/codec';
import { f32, techniqueProgram } from '@pmndrs/glyph/config/codec-program';
import { registerRasterPlanProgram } from '@pmndrs/glyph/config/raster';
import { defineTechniqueSchema } from '@pmndrs/glyph/config/schema';

interface FrameOptions {
  readonly inset?: number;
}
interface FrameDescriptor {
  readonly [key: string]: string | number;
  readonly generatorVersion: '0.0.0';
  readonly inset: number;
}
interface FrameData {
  readonly resource: RasterResourceId;
  readonly inset: number;
  readonly colors: Uint8Array;
  readonly glyphCount: number;
}
declare const frame: RasterFormat<RasterFormatId & 'studio.frame', 'frame', FrameOptions | undefined, FrameDescriptor, FrameData>;
declare const unitQuad: PortableGeometryPayload; // the example's 92-byte indexed unit quad

export const frameSchema = defineTechniqueSchema({
  technique: frame.id,
  scope: 'glyph',
  binding: { f32: ['inset', 'red', 'green', 'blue', 'alpha'] }, // per-glyph values from the decoded records
  buffers: {
    origin: { id: id.buffer('studio.frame/origin'), scalar: 'f32', lanes: ['left', 'top'] },
    size: { id: id.buffer('studio.frame/size'), scalar: 'f32', lanes: ['widthX', 'heightY'] },
    color: { id: id.buffer('studio.frame/color'), scalar: 'f32', lanes: ['red', 'green', 'blue', 'alpha'] },
  },
  resources: {
    glyphGeometry: {
      kind: 'geometry',
      attributes: [
        { semantic: 'position', componentType: 'f32', components: 3 },
        { semantic: 'uv', componentType: 'f32', components: 2 },
      ],
    },
  },
  render: { resource: 'glyphGeometry', geometry: { kind: 'quad', resource: 'glyphGeometry', coordinates: 'unit-square' } },
  glyphOrigin: { buffer: 'origin' }, // lanes 0–1 of `origin` are the glyph origin
});

export const framePlanProgram = registerRasterPlanProgram({
  raster: frame,
  schema: frameSchema,
  programVariant: 0,
  codecBody(system) {
    const p = techniqueProgram(frameSchema, { system });
    const { inlineOrigin, blockOrigin, fontSize, color } = p.semantics;
    const { inset, red, green, blue, alpha } = p.binding;
    const insetPixels = f32.mul(inset, fontSize);
    const twice = f32.mul(insetPixels, f32.const(2)); // the inset trims both edges
    return p.compile({
      origin: [f32.add(inlineOrigin, insetPixels), f32.sub(blockOrigin, insetPixels)],
      size: [f32.sub(f32.mul(fontSize, f32.const(0.65)), twice), f32.sub(fontSize, twice)],
      color: [f32.mul(color.red, red), f32.mul(color.green, green), f32.mul(color.blue, blue), f32.mul(color.alpha, alpha)],
    });
  },
  compileFont(compiler) {
    const data = compiler.font.data;
    compiler.retain('glyphGeometry', data.resource, unitQuad); // one supplied quad for every glyph
    return compiler.compile({
      strikes: [0], // one scalable strike
      resource: () => data.resource,
      f32: {
        inset: () => data.inset,
        red: (row) => data.colors[row * 4]! / 255,
        green: (row) => data.colors[row * 4 + 1]! / 255,
        blue: (row) => data.colors[row * 4 + 2]! / 255,
        alpha: (row) => data.colors[row * 4 + 3]! / 255,
      },
    });
  },
});
```

> [!WARNING]
> Register before the first handle.
